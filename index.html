Helper to gather all unique completion dates across all habits getCompletionDates() { // Fallback to checking local storage safely if your global variable differs const habits = JSON.parse(localStorage.getItem('habits')) || []; const allDates = new Set(); habits.forEach(habit => { if (habit.history && Array.isArray(habit.history)) { habit.history.forEach(dateStr => { // Extract just the YYYY-MM-DD part allDates.add(dateStr.split('T')[0]); }); } }); return allDates; }, // 2. Generate data blocks based on selected view timeline getTimelineData() { const completedDates = this.getCompletionDates(); const blocks = []; const now = new Date(); let totalCycles = 30; // Default Daily if (this.currentView === 'weekly') totalCycles = 12; if (this.currentView === 'monthly') totalCycles = 12; if (this.currentView === 'yearly') totalCycles = 365; for (let i = totalCycles - 1; i >= 0; i--) { let targetDate = new Date(); let label = ''; let isActive = false; if (this.currentView === 'daily' || this.currentView === 'yearly') { targetDate.setDate(now.getDate() - i); const dateKey = targetDate.toISOString().split('T')[0]; isActive = completedDates.has(dateKey); label = targetDate.toLocaleDateString(undefined, { month: 'short', day: 'numeric' }); } else if (this.currentView === 'weekly') { targetDate.setDate(now.getDate() - (i * 7)); // Check if any day in that entire week was active for (let d = 0; d < 7; d++) { const checkDate = new Date(targetDate); checkDate.setDate(targetDate.getDate() - d); if (completedDates.has(checkDate.toISOString().split('T')[0])) { isActive = true; break; } } label = `Wk -${i}`; } else if (this.currentView === 'monthly') { targetDate.setMonth(now.getMonth() - i); const targetMonth = targetDate.getMonth(); const targetYear = targetDate.getFullYear(); // Check if any day in that month was active completedDates.forEach(dateStr => { const d = new Date(dateStr); if (d.getMonth() === targetMonth && d.getFullYear() === targetYear) { isActive = true; } }); label = targetDate.toLocaleDateString(undefined, { month: 'short' }); } blocks.push({ isActive, label }); } return blocks; }, // 3.
